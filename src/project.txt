`default_nettype none

// UART Receiver Module
module uart_rx #(
  // parameter CLKS_PER_BIT = 10  // For 9600 baud rate with 100KHz clock
  parameter CLKS_PER_BIT = 10417
)(
  input wire clk,
  input wire rst_n,
  input wire rx,
  output reg rx_done,
  output reg [7:0] rx_data
);
  // State definitions
  localparam IDLE = 2'b00;
  localparam START = 2'b01;
  localparam DATA = 2'b10;
  localparam STOP = 2'b11;

  reg [1:0] state;
  reg [3:0] bit_counter;
  reg [$clog2(CLKS_PER_BIT)-1:0] clk_counter;
  
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      state <= IDLE;
      bit_counter <= 0;
      clk_counter <= 0;
      rx_done <= 0;
      rx_data <= 0;
    end else begin
      case (state)
        IDLE: begin
          rx_done <= 0;
          bit_counter <= 0;
          clk_counter <= 0;
          if (rx == 1'b0) begin  // Start bit detected
            state <= START;
          end
        end
        
        START: begin
          // Wait half bit period to sample in the middle of the bit
          if (clk_counter == (CLKS_PER_BIT-1)/2) begin
            if (rx == 1'b0) begin  // Confirm start bit
              clk_counter <= 0;
              state <= DATA;
            end else begin
              state <= IDLE;  // False start bit
            end
          end else begin
            clk_counter <= clk_counter + 1;
          end
        end
        
        DATA: begin
          if (clk_counter < CLKS_PER_BIT-1) begin
            clk_counter <= clk_counter + 1;
          end else begin
            clk_counter <= 0;
            rx_data <= {rx, rx_data[7:1]};  // LSB first
            
            if (bit_counter < 7) begin
              bit_counter <= bit_counter + 1;
            end else begin
              bit_counter <= 0;
              state <= STOP;
            end
          end
        end
        
        STOP: begin
          if (clk_counter < CLKS_PER_BIT-1) begin
            clk_counter <= clk_counter + 1;
          end else begin
            rx_done <= 1'b1;
            clk_counter <= 0;
            state <= IDLE;
          end
        end
        
        default: state <= IDLE;
      endcase
    end
  end
endmodule

// UART Transmitter Module
module uart_tx #(
  // parameter CLKS_PER_BIT = 10  // For 9600 baud rate with 100KHz clock
  parameter CLKS_PER_BIT = 10417
)(
  input wire clk,
  input wire rst_n,
  input wire tx_start,
  input wire [7:0] tx_data,
  output reg tx,
  output reg tx_done
);
  // State definitions
  localparam IDLE = 2'b00;
  localparam START = 2'b01;
  localparam DATA = 2'b10;
  localparam STOP = 2'b11;
  
  reg [1:0] state;
  reg [2:0] bit_counter;
  reg [$clog2(CLKS_PER_BIT)-1:0] clk_counter;
  reg [7:0] tx_data_reg;
  
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      state <= IDLE;
      bit_counter <= 0;
      clk_counter <= 0;
      tx <= 1'b1;  // Idle state is high
      tx_done <= 0;
      tx_data_reg <= 0;
    end else begin
      case (state)
        IDLE: begin
          tx <= 1'b1;  // Idle state is high
          tx_done <= 0;
          bit_counter <= 0;
          clk_counter <= 0;
          
          if (tx_start) begin
            tx_data_reg <= tx_data;
            state <= START;
          end
        end
        
        START: begin
          tx <= 1'b0;  // Start bit is low
          
          if (clk_counter < CLKS_PER_BIT-1) begin
            clk_counter <= clk_counter + 1;
          end else begin
            clk_counter <= 0;
            state <= DATA;
          end
        end
        
        DATA: begin
          tx <= tx_data_reg[bit_counter];  // LSB first
          
          if (clk_counter < CLKS_PER_BIT-1) begin
            clk_counter <= clk_counter + 1;
          end else begin
            clk_counter <= 0;
            
            if (bit_counter < 7) begin
              bit_counter <= bit_counter + 1;
            end else begin
              bit_counter <= 0;
              state <= STOP;
            end
          end
        end
        
        STOP: begin
          tx <= 1'b1;  // Stop bit is high
          
          if (clk_counter < CLKS_PER_BIT-1) begin
            clk_counter <= clk_counter + 1;
          end else begin
            tx_done <= 1'b1;
            clk_counter <= 0;
            state <= IDLE;
          end
        end
        
        default: state <= IDLE;
      endcase
    end
  end
endmodule

// Original Matrix Multiplication Module (now as a submodule)
module matrix_multiplier (
    input  wire [7:0] ui_in,    // Input for first matrix values
    output wire [7:0] uo_out,   // Sequential output of result cells
    input  wire [7:0] uio_in,   // Input for second matrix values
    output wire [7:0] uio_out,  // (Unused)
    output wire [7:0] uio_oe,   // (Unused)
    input  wire       ena,      // always 1 when design powered
    input  wire       clk,      // clock
    input  wire       rst_n     // active-low reset
);

  // Unused outputs fixed to 0.
  assign uio_out = 0;
  assign uio_oe  = 0;

  // Internal state definitions
  localparam STATE_INPUT_A = 2'd0,
             STATE_INPUT_B = 2'd1,
             STATE_COMPUTE = 2'd2,
             STATE_OUTPUT  = 2'd3;

  reg [1:0] state;
  reg [1:0] counter;      // used for both input and output counting
  reg [7:0] A0, A1, A2, A3;   // Elements for matrix A (rowwise order)
  reg [7:0] B0, B1, B2, B3;   // Elements for matrix B (rowwise order)
  reg [15:0] C00, C01, C10, C11; // Full-precision intermediate products
  reg [7:0] uo_out_reg;       // Registered output (truncated to 8 bits)

  // Output assignment
  assign uo_out = uo_out_reg;

  // Sequential logic with state machine
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      state      <= STATE_INPUT_A;
      counter    <= 0;
      A0 <= 0; A1 <= 0; A2 <= 0; A3 <= 0;
      B0 <= 0; B1 <= 0; B2 <= 0; B3 <= 0;
      C00 <= 0; C01 <= 0; C10 <= 0; C11 <= 0;
      uo_out_reg <= 0;
    end else begin
      case (state)
        // Input first matrix (from ui_in)
        STATE_INPUT_A: begin
          case (counter)
            2'd0: A0 <= ui_in;
            2'd1: A1 <= ui_in;
            2'd2: A2 <= ui_in;
            2'd3: A3 <= ui_in;
          endcase
          if (counter == 2'd3) begin
            state   <= STATE_INPUT_B;
            counter <= 0;
          end else begin
            counter <= counter + 1;
          end
        end

        // Input second matrix (from uio_in)
        STATE_INPUT_B: begin
          case (counter)
            2'd0: B0 <= uio_in;
            2'd1: B1 <= uio_in;
            2'd2: B2 <= uio_in;
            2'd3: B3 <= uio_in;
          endcase
          if (counter == 2'd3) begin
            state   <= STATE_COMPUTE;
            counter <= 0;
          end else begin
            counter <= counter + 1;
          end
        end

        // Compute matrix multiplication
        STATE_COMPUTE: begin
          // Standard 2x2 matrix multiply (results are truncated to 8 bits)
          C00 <= A0 * B0 + A1 * B2;
          C01 <= A0 * B1 + A1 * B3;
          C10 <= A2 * B0 + A3 * B2;
          C11 <= A2 * B1 + A3 * B3;
          state <= STATE_OUTPUT;
          counter <= 0;
        end

        // Output each cell over successive clock cycles
        STATE_OUTPUT: begin
          case (counter)
            2'd0: uo_out_reg <= C00[7:0];
            2'd1: uo_out_reg <= C01[7:0];
            2'd2: uo_out_reg <= C10[7:0];
            2'd3: uo_out_reg <= C11[7:0];
          endcase
          if (counter == 2'd3) begin
            // After outputting all cells, restart input (or hold, depending on design needs)
            state   <= STATE_INPUT_A;
            counter <= 0;
          end else begin
            counter <= counter + 1;
          end
        end

        default: state <= STATE_INPUT_A;
      endcase
    end
  end

  // List all unused input signals to prevent warnings
  wire _unused = &{ena};

endmodule

// Top-level UART Matrix Multiplier
module tt_um_2x2MatrixMult_Vort3xed (
    input  wire [7:0] ui_in,    // ui_in[0] is used for RX pin
    output wire [7:0] uo_out,   // uo_out[0] is used for TX pin
    input  wire [7:0] uio_in,   // Unused
    output wire [7:0] uio_out,  // Unused
    output wire [7:0] uio_oe,   // Unused
    input  wire       ena,      // always 1 when design powered
    input  wire       clk,      // clock
    input  wire       rst_n     // active-low reset
);
  // Define UART baud rate
  // parameter CLKS_PER_BIT = 10;  // For 9600 baud with 100KHz clock
  parameter CLKS_PER_BIT = 10417

  // State machine definitions
  localparam WAIT_MATRIX_A = 3'd0,
             WAIT_MATRIX_B = 3'd1,
             COMPUTE      = 3'd2,
             SEND_RESULTS = 3'd3,
             DONE         = 3'd4;
  
  reg [2:0] state;
  reg [2:0] byte_counter;
  
  // UART signals
  wire rx = ui_in[0];           // UART RX on the first bit of ui_in
  wire tx;                      // UART TX wire
  assign uo_out = {7'b0, tx};   // UART TX on the first bit of uo_out
  
  // UART RX signals
  wire rx_done;
  wire [7:0] rx_data;
  
  // UART TX signals
  reg tx_start;
  reg [7:0] tx_data;
  wire tx_done;
  
  // Matrix multiplier signals
  reg [7:0] mm_ui_in;
  reg [7:0] mm_uio_in;
  wire [7:0] mm_uo_out;
  
  // Matrix storage
  reg [7:0] matrix_A [0:3];
  reg [7:0] matrix_B [0:3];
  reg [7:0] result_matrix [0:3];
  
  // Instantiate UART RX
  uart_rx #(
    .CLKS_PER_BIT(CLKS_PER_BIT)
  ) rx_inst (
    .clk(clk),
    .rst_n(rst_n),
    .rx(rx),
    .rx_done(rx_done),
    .rx_data(rx_data)
  );
  
  // Instantiate UART TX
  uart_tx #(
    .CLKS_PER_BIT(CLKS_PER_BIT)
  ) tx_inst (
    .clk(clk),
    .rst_n(rst_n),
    .tx_start(tx_start),
    .tx_data(tx_data),
    .tx(tx),
    .tx_done(tx_done)
  );
  
  // Instantiate original matrix multiplier
  matrix_multiplier mm_inst (
    .ui_in(mm_ui_in),
    .uo_out(mm_uo_out),
    .uio_in(mm_uio_in),
    .uio_out(),  // Unused
    .uio_oe(),   // Unused
    .ena(ena),
    .clk(clk),
    .rst_n(rst_n)
  );
  
  // Main state machine for UART protocol
  always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
      state <= WAIT_MATRIX_A;
      byte_counter <= 0;
      tx_start <= 0;
      tx_data <= 0;
      mm_ui_in <= 0;
      mm_uio_in <= 0;
    end else begin
      // Default values
      tx_start <= 0;
      
      case (state)
        // Receiving Matrix A
        WAIT_MATRIX_A: begin
          if (rx_done) begin
            matrix_A[byte_counter] <= rx_data;
            if (byte_counter == 3) begin
              byte_counter <= 0;
              state <= WAIT_MATRIX_B;
            end else begin
              byte_counter <= byte_counter + 1;
            end
          end
        end
        
        // Receiving Matrix B
        WAIT_MATRIX_B: begin
          if (rx_done) begin
            matrix_B[byte_counter] <= rx_data;
            if (byte_counter == 3) begin
              byte_counter <= 0;
              state <= COMPUTE;
            end else begin
              byte_counter <= byte_counter + 1;
            end
          end
        end
        
        // Feed values to matrix multiplier
        COMPUTE: begin
          case (byte_counter)
            // Feed Matrix A
            3'd0: begin mm_ui_in <= matrix_A[0]; byte_counter <= byte_counter + 1; end
            3'd1: begin mm_ui_in <= matrix_A[1]; byte_counter <= byte_counter + 1; end
            3'd2: begin mm_ui_in <= matrix_A[2]; byte_counter <= byte_counter + 1; end
            3'd3: begin mm_ui_in <= matrix_A[3]; byte_counter <= byte_counter + 1; end
            
            // Feed Matrix B
            3'd4: begin mm_uio_in <= matrix_B[0]; byte_counter <= byte_counter + 1; end
            3'd5: begin mm_uio_in <= matrix_B[1]; byte_counter <= byte_counter + 1; end
            3'd6: begin mm_uio_in <= matrix_B[2]; byte_counter <= byte_counter + 1; end
            3'd7: begin mm_uio_in <= matrix_B[3]; byte_counter <= 0; state <= SEND_RESULTS; end
          endcase
        end
        
        // Collect and send results
        SEND_RESULTS: begin
          case (byte_counter)
            // Collect results
            3'd0: begin result_matrix[0] <= mm_uo_out; byte_counter <= byte_counter + 1; end
            3'd1: begin result_matrix[1] <= mm_uo_out; byte_counter <= byte_counter + 1; end
            3'd2: begin result_matrix[2] <= mm_uo_out; byte_counter <= byte_counter + 1; end
            3'd3: begin result_matrix[3] <= mm_uo_out; byte_counter <= byte_counter + 1; end
            
            // Send results over UART
            3'd4: begin
              if (!tx_start && !tx_done) begin
                tx_data <= result_matrix[0];
                tx_start <= 1;
              end else if (tx_done) begin
                byte_counter <= byte_counter + 1;
              end
            end
            3'd5: begin
              if (!tx_start && !tx_done) begin
                tx_data <= result_matrix[1];
                tx_start <= 1;
              end else if (tx_done) begin
                byte_counter <= byte_counter + 1;
              end
            end
            3'd6: begin
              if (!tx_start && !tx_done) begin
                tx_data <= result_matrix[2];
                tx_start <= 1;
              end else if (tx_done) begin
                byte_counter <= byte_counter + 1;
              end
            end
            3'd7: begin
              if (!tx_start && !tx_done) begin
                tx_data <= result_matrix[3];
                tx_start <= 1;
              end else if (tx_done) begin
                byte_counter <= 0;
                state <= DONE;
              end
            end
          endcase
        end
        
        // Done, ready for next operation
        DONE: begin
          state <= WAIT_MATRIX_A;  // Ready for next matrix
        end
        
        default: state <= WAIT_MATRIX_A;
      endcase
    end
  end
  
  // Tie off unused output pins
  assign uio_out = 8'b0;
  assign uio_oe = 8'b0;

endmodule
